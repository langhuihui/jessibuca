📖 使用示例

1. HLS 播放

<script type="module">
  import 'jv4-ui';
</script>

<jessibuca
  src="https://example.com/video.m3u8"
  show-playback-rate
  show-progress
  auto-generate-ui
></jessibuca>

2. WebSocket FLV 播放

<jessibuca
  src="ws://example.com/live/stream.flv"
  auto-generate-ui
></jessibuca>

3. WebRTC 播放

<jessibuca
  src="webrtc://example.com/live/stream"
  auto-generate-ui
></jessibuca>

4. 带调试模式

<jessibuca
  src="https://example.com/video.m3u8"
  debug
  show-time-ranges
  show-media-timeline
></jessibuca>

3. JavaScript 控制

const player = document.querySelector('jessibuca');

// 设置视频源
player.src = 'https://example.com/video.m3u8';

// 播放控制
await player.play();
player.pause();
player.seek(30);

// 获取状态
const currentTime = player.getCurrentTime();
const duration = player.getDuration();

// 监听事件
player.addEventListener('play', () => console.log('开始播放'));
player.addEventListener('pause', () => console.log('暂停播放'));
player.addEventListener('timeupdate', (e) => {
  console.log('时间更新:', e.detail);
});
